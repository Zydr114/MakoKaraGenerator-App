cmake_minimum_required(VERSION 3.20)
project(mkg_envelope_probe LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The native envelope acceptance fixture requires Linux")
endif()
add_executable(mkg-envelope-probe-helper mkg-envelope-probe-helper.c)
target_compile_features(mkg-envelope-probe-helper PRIVATE c_std_11)
target_compile_options(mkg-envelope-probe-helper PRIVATE -Wall -Wextra -Werror)
install(TARGETS mkg-envelope-probe-helper RUNTIME DESTINATION .)
