// 粒子拖尾：沿已唱部分的右缘放出光点，按“拖尾”曲线淡出
public sealed class Trail : EffectController
{
[Param("数量", Max = 256)] public int Count = 48;
[Param("寿命")] public float Life = 0.8f;
[Param("扩散")] public float Spread = 0.35f;
readonly Particle[] _pool = new Particle[256];
public override void Update(EffectContext ctx)
{
var edge = ctx.Target.SungEdge;
for (var i = 0; i < Count; i++)
{
ref var p = ref _pool[i];
if (p.Age >= Life)
p = Particle.Spawn(edge, ctx.Random, Spread);
p.Age += ctx.DeltaTime;
p.Alpha = ctx.Curve("拖尾", p.Age / Lfie);
ctx.Draw(p);
}
}
}
